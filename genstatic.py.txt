def chg(f):
    if len(ls)<2: return 'First snapshot'
    dd=ls[-1][f]-ls[-2][f]; return 'No change' if dd==0 else f"{'Up' if dd>0 else 'Down'} {abs(dd)} pts since {e(ls[-2]['label'])}"
def table(head,rows,num=(),cls=None):
    h=''.join(f'<th{" class=n" if i in num else ""}>{x}</th>' for i,x in enumerate(head))
    b=''.join((f'<tr class="{cls(i)}">' if cls and cls(i) else '<tr>')+''.join(f'<td{" class=n" if j in num else ""}>{e(x) if not str(x).startswith("&lt;") else x}</td>' for j,x in enumerate(r))+'</tr>' for i,r in enumerate(rows))
    return f'<div class="wrap"><table><thead><tr>{h}</tr></thead><tbody>{b}</tbody></table></div>'

# scatter

X=lambda v: m['l']+(SW-m['l']-m['r'])*v/xm; Y=lambda v: m['t']+(SH-m['t']-m['b'])*(ymax-min(v,ymax))/(ymax-ymin)
g=''.join(f'<line x1="{m["l"]}" x2="{SW-m["r"]}" y1="{Y(t):.1f}" y2="{Y(t):.1f}" class="{"zero" if t==0 else "grid"}"/><text x="{m["l"]-8}" y="{Y(t)+4:.1f}" text-anchor="end">{t}</text>' for t in range(-20,71,10) if t%20==0)
g+=''.join(f'<text x="{X(t):.1f}" y="{SH-m["b"]+16}" text-anchor="middle">{t}</text>' for t in range(0,61,10))
g+=f'<line x1="{X(0)}" y1="{Y(0)}" x2="{X(60)}" y2="{Y(60)}" class="ref"/><text x="{(m["l"]+SW-m["r"])/2}" y="{SH-6}" text-anchor="middle">Predicted margin for the favorite</text><text transform="translate(12,{(SH-m["b"]+m["t"])/2}) rotate(-90)" text-anchor="middle">Actual margin</text>'
for r in results:
    me='Lindale' in (r['favorite'],r['underdog'])
    g+=f'<circle cx="{X(r["pred_margin"]):.1f}" cy="{Y(r["actual_margin"]):.1f}" r="{7 if me else 5}" class="{"won" if r["fav_won"] else "ups"}{" me" if me else ""}"><title>{e(r["winner"])} {e(r["score"])} {e(r["loser"])}. {e(r["favorite"])} was favored by {r["pred_margin"]:g}.</title></circle>'
    if r['actual_margin']>ymax: g+=f'<text x="{X(r["pred_margin"])+9:.1f}" y="{Y(ymax)+4:.1f}">won by {r["actual_margin"]}</text>'
    if me: g+=f'<text x="{X(r["pred_margin"])+11:.1f}" y="{Y(r["actual_margin"])+4:.1f}" class="val">Lindale</text>'
scsvg=f'<svg viewBox="0 0 {SW} {SH}" role="img" aria-label="Predicted versus actual margin for the favorite in each playoff game">{g}</svg>'

    for k in sorted(set(r['slot']//size for r in reg)):

    return out+'</div>'
