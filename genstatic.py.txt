def chg(f):
    if len(ls)<2: return ''
    dd=ls[-1][f]-ls[-2][f]; return 'No change' if dd==0 else f"{'Up' if dd>0 else 'Down'} {abs(dd)} pts since {e(ls[-2]['label'])}"
def table(head,rows,num=(),cls=None,raw=(),wrapc=(),stack=False,hm=()):
    lab=lambda x: html.escape(__import__('re').sub('<[^>]+>','',x),quote=True)
    h=''.join('<th'+(' class="'+' '.join(k for k,v in (('n',i in num),('hm',i in hm)) if v)+'"' if (i in num or i in hm) else '')+f'>{x}</th>' for i,x in enumerate(head))
    def td(j,x):
        c=' '.join(k for k,v in (('n',j in num),('w',j in wrapc),('hm',j in hm)) if v)
        t=x if (j in raw or str(x).startswith("&lt;")) else e(x)
        return f'<td{" class=\""+c+"\"" if c else ""} data-label="{lab(head[j])}">{t}</td>'
    b=''.join((f'<tr class="{cls(i)}">' if cls and cls(i) else '<tr>')+''.join(td(j,x) for j,x in enumerate(r))+'</tr>' for i,r in enumerate(rows))
    return f'<div class="wrap{" stack" if stack else ""}"><table><thead><tr>{h}</tr></thead><tbody>{b}</tbody></table></div>'
def pathrow(r):
    opp=e(r['opponent'])+(f"<small>{e(r['earlier'])}</small>" if r['earlier'] else '')
    bits=[e(r['site_status'])]+([f"{r['capacity']:,} seats"] if r['capacity'] else [])+([f"{r['drive_miles']} mi"] if r['drive_miles'] else [])+([e(r['drive_time'])] if r['drive_time'] else [])
    site=e(r['site'])+f"<small>{' · '.join(bits)}</small>"
    return [r['round'],r['when'],opp,f"{r['opp_share']}%",f"{r['win_prob']}%",site]
